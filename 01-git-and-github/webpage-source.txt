let lastX=0,lastY=0,startX=0,startY=0,lastTime=0,lastMoveTime=0,raf=0;
// 缓存固定三角函数和投影缓冲区，避免每帧创建几千个对象。
const points=[];
for(let row=0;row<92;row++)for(let col=0;col<23;col++){
  const u=row/92*Math.PI*2,v=col/23*Math.PI*2;
  points.push({cu:Math.cos(u),su:Math.sin(u),s3:Math.sin(u*3),c3:Math.cos(u*3),
    cv:Math.cos(v),sv:Math.sin(v),spread:.4+Math.abs(Math.sin(row*127.1+col*311.7)*43758.5453%1)*1.5});
}
const px=new Float32Array(points.length),py=new Float32Array(points.length),sizes=new Float32Array(points.length);
const buckets=Array.from({length:16},()=>[]);
function render() {
  if(!ctx)return;
  ctx.clearRect(0,0,width,height);
  const scale=Math.min(width,height)*.27;
  const ca=Math.cos(angle),sa=Math.sin(angle),ct=Math.cos(tilt),st=Math.sin(tilt);
  const color=palettes[paletteIndex].rgb.join(",");
  const pointScale=Math.min(width/500,1.25);
  for(const bucket of buckets)bucket.length=0;
  // 同一环截面的扭转参数只计算一次。
  let co=1,si=0,wave=0;
  for(let i=0;i<points.length;i++){
    const p=points[i];
    if(i%23===0){const offset=p.s3*freedom*1.8;co=Math.cos(offset);si=Math.sin(offset);wave=(p.s3*ca+p.c3*sa)*freedom*.38;}
    const cosTwist=p.cv*co-p.sv*si,sinTwist=p.sv*co+p.cv*si;
    const radius=1.08+.43*cosTwist,expand=1+pulse*p.spread;
    const x=radius*p.cu*expand,y=(.43*sinTwist+wave)*expand,z=radius*p.su*expand;
    const rx=x*ca-z*sa,rz=x*sa+z*ca,ry=y*ct-rz*st,depth=y*st+rz*ct;
    const perspective=3.7/Math.max(.8,3.7+depth);
    px[i]=width*.5+rx*scale*perspective;py[i]=height*.48+ry*scale*perspective;
    sizes[i]=(1+perspective*.32)*pointScale;
    const shade=Math.max(0,Math.min(15,Math.floor((1.7-depth)/3.7*16)));
    buckets[shade].push(i);
  }
  // 同色粒子按深度分为 16 批绘制，替代逐点排序和两千次 fill。
  for(let shade=0;shade<buckets.length;shade++){
    ctx.beginPath();
    for(const i of buckets[shade]){ctx.moveTo(px[i]+sizes[i],py[i]);ctx.arc(px[i],py[i],sizes[i],0,Math.PI*2);}
    ctx.fillStyle="rgba("+color+","+(.22+shade/15*.68)+")";ctx.fill();
  }
}
function resize(){
  const rect=canvas.getBoundingClientRect();width=rect.width;height=rect.height;
  const dpr=Math.min(window.devicePixelRatio||1,2);
  canvas.width=Math.round(width*dpr);canvas.height=Math.round(height*dpr);
  if(ctx)ctx.setTransform(dpr,0,0,dpr,0,0);render();
}
function updatePause(){ $("#pause").textContent=paused?"继续流动":"暂停流动";$("#pause").setAttribute("aria-pressed",String(paused));}
function settleImmediately(){
  angle=targetAngle;tilt=targetTilt;freedom=targetFreedom;
  pulse=pulseTarget=0;velocityX=velocityY=0;render();
}
function needsFrame(){
  return !paused||dragging||Math.abs(targetAngle-angle)>.0001||Math.abs(targetTilt-tilt)>.0001||
    Math.abs(targetFreedom-freedom)>.0001||Math.abs(velocityX)+Math.abs(velocityY)>.001||pulse>.0001||pulseTarget>.0001;
}
function frame(time){
  raf=0;
  if(!visible)return;
  const dt=Math.min(Math.max((time-lastTime)/1000,0),.05);lastTime=time;
  if(!paused&&!dragging)targetAngle+=dt*.19;
  if(!dragging&&!reduced.matches){
    targetAngle+=velocityX*dt;
    targetTilt=Math.max(-1.4,Math.min(1.4,targetTilt+velocityY*dt));
    const friction=Math.exp(-5.5*dt);velocityX*=friction;velocityY*=friction;
    if(Math.abs(velocityX)<.001)velocityX=0;
    if(Math.abs(velocityY)<.001)velocityY=0;
  }
  // 按真实时间阻尼，让 60 Hz 和高刷新率屏幕具有接近的手感。
  const follow=1-Math.exp(-(dragging?32:15)*dt);
  angle+=(targetAngle-angle)*follow;tilt+=(targetTilt-tilt)*follow;
  freedom+=(targetFreedom-freedom)*(1-Math.exp(-12*dt));
  pulse+=(pulseTarget-pulse)*(1-Math.exp(-22*dt));
  pulseTarget*=Math.exp(-3.8*dt);
  if(pulse<.0001&&pulseTarget<.0001)pulse=pulseTarget=0;
  render();
  if(needsFrame())raf=requestAnimationFrame(frame);
}
function start(){
  if(!visible)return;
  if(reduced.matches){settleImmediately();if(paused)return;}
  if(!raf){lastTime=performance.now();raf=requestAnimationFrame(frame);}
}
function burst(){
  if(reduced.matches){targetAngle+=.6;settleImmediately();notify("换个角度，可能不止一种。");return;}
  pulseTarget=Math.min(.9,pulseTarget+.8);start();
}

  targetAngle=angle+Math.atan2(Math.sin(.35-angle),Math.cos(.35-angle));
  targetTilt=.55;targetFreedom=.28;pulseTarget=0;velocityX=velocityY=0;paused=reduced.matches;
  $("#energy").value="28";$("#energy-value").value="28";setPalette(0,false);updatePause();start();notify("回到最初，再试一种可能。");
});
canvas.addEventListener("pointerdown",e=>{
  if(e.button!==0||activePointer!==null)return;
  activePointer=e.pointerId;dragging=true;moved=false;velocityX=velocityY=0;
  targetAngle=angle;targetTilt=tilt;
  lastX=startX=e.clientX;lastY=startY=e.clientY;lastMoveTime=performance.now();
  canvas.setPointerCapture(e.pointerId);canvas.classList.add("dragging");start();
});
canvas.addEventListener("pointermove",e=>{
  if(!dragging||e.pointerId!==activePointer)return;
  const now=performance.now(),dt=Math.max((now-lastMoveTime)/1000,.008);
  const dx=(e.clientX-lastX)*.006,dy=(e.clientY-lastY)*.006;
  if(Math.hypot(e.clientX-startX,e.clientY-startY)>6)moved=true;
  targetAngle+=dx;targetTilt=Math.max(-1.4,Math.min(1.4,targetTilt+dy));
  velocityX=velocityX*.4+Math.max(-3,Math.min(3,dx/dt))*.6;
  velocityY=velocityY*.4+Math.max(-2,Math.min(2,dy/dt))*.6;
  lastX=e.clientX;lastY=e.clientY;lastMoveTime=now;
  // 输入事件只更新目标；统一在动画帧中绘制。
  start();
});
function endDrag(e){
  if(!dragging||e.pointerId!==activePointer)return;
  dragging=false;activePointer=null;canvas.classList.remove("dragging");
  if(canvas.hasPointerCapture(e.pointerId))canvas.releasePointerCapture(e.pointerId);
  if(e.type!=="pointerup"||!moved||paused||reduced.matches||performance.now()-lastMoveTime>100)velocityX=velocityY=0;
  if(e.type==="pointerup"&&!moved)burst();else start();
}
canvas.addEventListener("pointerup",endDrag);canvas.addEventListener("pointercancel",endDrag);
canvas.addEventListener("lostpointercapture",endDrag);
canvas.addEventListener("keydown",e=>{
  const actions={ArrowLeft:()=>targetAngle-=.15,ArrowRight:()=>targetAngle+=.15,ArrowUp:()=>targetTilt=Math.max(-1.4,targetTilt-.15),ArrowDown:()=>targetTilt=Math.min(1.4,targetTilt+.15)," ":burst};
  if(actions[e.key]){e.preventDefault();actions[e.key]();start();}
});
function syncVisibility(){
  visible=inView&&!document.hidden;
  if(visible)start();
  else{
    cancelAnimationFrame(raf);raf=0;velocityX=velocityY=0;
    if(activePointer!==null&&canvas.hasPointerCapture(activePointer))canvas.releasePointerCapture(activePointer);
    activePointer=null;dragging=false;canvas.classList.remove("dragging");
  }
}

function wrapText(context,text,maxWidth){
  const lines=[];for(const paragraph of text.split("\n")){
    let line="";for(const char of paragraph){if(line&&context.measureText(line+char).width>maxWidth){lines.push(line);line=char;}else line+=char;}lines.push(line);
  }return lines;
}
